  const getLockoutKey = (p) => `login_lock_${(p || '').trim()}`;

  const checkLockout = (p) => {
    try {
      const raw = localStorage.getItem(getLockoutKey(p));
      if (!raw) return { isLocked: false, remainingSec: 0, attempts: 0 };
      const data = JSON.parse(raw);
      if (data.lockedUntil && Date.now() < data.lockedUntil) {
        const remainingSec = Math.ceil((data.lockedUntil - Date.now()) / 1000);
        return { isLocked: true, remainingSec, attempts: data.attempts || 5 };
      }
      if (data.lockedUntil && Date.now() >= data.lockedUntil) {
        localStorage.removeItem(getLockoutKey(p));
      }
      return { isLocked: false, remainingSec: 0, attempts: data.attempts || 0 };
    } catch(e) {
      return { isLocked: false, remainingSec: 0, attempts: 0 };
    }
  };

  const handleLogin = async (e) => {
    e.preventDefault();

    const lockStatus = checkLockout(phone);
    if (lockStatus.isLocked) {
      const mins = Math.ceil(lockStatus.remainingSec / 60);
      const msg = lang === 'en'
        ? `⚠️ Account is temporarily locked after 5 failed attempts. Please wait ${mins} minute(s) before trying again.`
        : `⚠️ تم إيقاف محاولات تسجيل الدخول مؤقتاً لحماية أمان حسابك بعد 5 محاولات غير صحيحة. يرجى الانتظار ${mins} دقيقة والمحاولة لاحقاً.`;
      setError(msg);
      alert(msg);
      return;
    }

    if (!password) return setError('يرجى إدخال كلمة المرور');
    setError('');
    setLoading(true);

    try {
      const custRef = ref(db, `PublicCustomers/${phone}`);
      const custSnap = await get(custRef);
      
      const internalEmail = getInternalEmail(phone);
      const dbEmail = custSnap.exists() && custSnap.val().Email ? custSnap.val().Email.trim() : null;
      const authEmail = custSnap.exists() && custSnap.val().AuthEmail ? custSnap.val().AuthEmail.trim() : null;
      
      const emailToTry1 = authEmail || internalEmail;
      
      try {
        // Try internal/auth email first (for accounts created with phone only or updated via reset)
        await signInWithEmailAndPassword(auth, emailToTry1, password);
      } catch (firstErr) {
        // If it fails, and they have a real email, try that (for accounts created with real email)
        if (dbEmail && dbEmail !== emailToTry1 && dbEmail.includes('@')) {
          try {
            await signInWithEmailAndPassword(auth, dbEmail, password);
          } catch (secondErr) {
            throw secondErr; // Both failed
          }
        } else {
          throw firstErr; // Only had first email, and it failed
        }
      }
      
      // Reset lockout on successful login
      try { localStorage.removeItem(getLockoutKey(phone)); } catch(e) {}
      await refreshCustomerData(phone);
      setStep('dashboard');
    } catch (err) {
      let updatedAttempts = 1;
      try {
        const key = getLockoutKey(phone);
        const raw = localStorage.getItem(key);
        const data = raw ? JSON.parse(raw) : { attempts: 0 };
        data.attempts = (data.attempts || 0) + 1;
        data.lastAttempt = Date.now();
        if (data.attempts >= 5) {
          data.lockedUntil = Date.now() + 5 * 60 * 1000; // Lock for 5 mins
          localStorage.setItem(key, JSON.stringify(data));
          const lockMsg = lang === 'en'
            ? '⚠️ You have entered an incorrect password 5 times. Account login is locked for 5 minutes.'
            : '⚠️ لقد قمت بإدخال كلمة مرور غير صحيحة 5 مرات متتالية. تم إيقاف محاولات الدخول مؤقتاً لمدة 5 دقائق لحماية أمان حسابك.';
          setError(lockMsg);
          alert(lockMsg);
          setLoading(false);
          return;
        } else {
          localStorage.setItem(key, JSON.stringify(data));
          updatedAttempts = data.attempts;
        }
      } catch(e) {}

      const remainingAttempts = Math.max(0, 5 - updatedAttempts);
      const errMsg = lang === 'en'
        ? `Incorrect password. Remaining attempts before temporary lockout: ${remainingAttempts}`
        : `رقم الهاتف أو كلمة المرور غير صحيحة. المحاولات المتبقية قبل إيقاف الحساب مؤقتاً: ${remainingAttempts}`;
      setError(errMsg);
    }
    setLoading(false);
  };
