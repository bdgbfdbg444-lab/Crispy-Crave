  const [email, setEmail] = useState('');
  const [newPassword, setNewPassword] = useState('');
  
  // WhatsApp OTP Authentication State
  const [step, setStep] = useState('phone');
  const [otpInput, setOtpInput] = useState('');
  const [generatedOtp, setGeneratedOtp] = useState('');
  const [otpExpiry, setOtpExpiry] = useState(0);
  const [otpCountdown, setOtpCountdown] = useState(0);
  const [isExistingCustomer, setIsExistingCustomer] = useState(false);
  const [existingCustomerRecord, setExistingCustomerRecord] = useState(null);

  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    // If auth state resolved and we have both user and data, go to dashboard
    if (!authLoading && currentUser && customerData) {
      setStep('dashboard');
    }
  }, [currentUser, customerData, authLoading]);

  // Countdown timer for OTP resend
  useEffect(() => {
    let timer;
    if (otpCountdown > 0) {
      timer = setInterval(() => {
        setOtpCountdown(prev => prev - 1);
      }, 1000);
    }
    return () => clearInterval(timer);
  }, [otpCountdown]);

  if (authLoading) return <div className="text-center mt-20">جاري التحميل...</div>;

  const getInternalEmail = (p) => `${p}@internal.theblackbox.com`;

  const normalizePhone = (p) => {
    let clean = (p || '').replace(/\D/g, '');
    return clean;
  };

  const sendWhatsAppOtp = async (targetPhone) => {
    const cleanPhone = normalizePhone(targetPhone);
    const code = Math.floor(100000 + Math.random() * 900000).toString();
    const expiry = Date.now() + 5 * 60 * 1000; // 5 minutes valid

    setGeneratedOtp(code);
    setOtpExpiry(expiry);
    setOtpCountdown(60); // 60s cooldown for resend

    // Write to Firebase PendingOtpRequests for silent POS WhatsApp Server
    const reqId = `otp_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`;
    await set(ref(db, `PendingOtpRequests/${reqId}`), {
      phone: cleanPhone,
      otp: code,
      createdAt: Date.now()
    });

    return code;
  };

  const handlePhoneSubmit = async (e) => {
    e.preventDefault();
    const cleanPhone = normalizePhone(phone);
    if (cleanPhone.length < 10) return setError('رقم الهاتف غير صحيح');
    setError('');
    setLoading(true);

    try {
      const custRef = ref(db, `PublicCustomers/${cleanPhone}`);
      const custSnap = await get(custRef);
      if (custSnap.exists()) {
        const data = custSnap.val();
        setIsExistingCustomer(true);
        setExistingCustomerRecord(data);
        if (data.uid) {
          // Has set a password before
          setStep('login');
        } else {
          // Registered in POS only, trigger WhatsApp OTP verification to verify and set password!
          await sendWhatsAppOtp(cleanPhone);
          setStep('otp_verify');
        }
      } else {
        setIsExistingCustomer(false);
        setExistingCustomerRecord(null);
        // Completely new web customer: Send WhatsApp OTP first to verify phone!
        await sendWhatsAppOtp(cleanPhone);
        setStep('otp_verify');
      }
    } catch (err) {
      setError('حدث خطأ أثناء فحص الرقم: ' + err.message);
    }
    setLoading(false);
  };

  const handleResendOtp = async () => {
    if (otpCountdown > 0) return;
    setLoading(true);
    try {
      await sendWhatsAppOtp(phone);
      setError('');
      alert(lang === 'en' ? 'A new verification code was sent to your WhatsApp.' : 'تم إرسال كود تحقق جديد إلى رقم واتساب الخاص بك.');
    } catch (err) {
      setError('فشل إرسال كود الواتساب: ' + err.message);
    }
    setLoading(false);
  };

  const handleVerifyOtp = async (e) => {
    e.preventDefault();
    if (!otpInput || otpInput.trim().length !== 6) {
      return setError('يرجى إدخال رمز التحقق المكون من 6 أرقام');
    }
    if (Date.now() > otpExpiry) {
      return setError('انتهت صلاحية رمز التحقق. يرجى طلب رمز جديد.');
    }
    if (otpInput.trim() !== generatedOtp) {
      return setError('رمز التحقق غير صحيح');
    }

    // OTP verified successfully!
    setError('');
    if (isExistingCustomer) {
      // Existing customer from POS, take to set their account password
      setStep('set_password');
    } else {
      // New customer, proceed to complete profile details
      setStep('complete_profile');
    }
  };

  const getLockoutKey = (p) => `login_lock_${(p || '').trim()}`;

  const checkLockout = (p) => {
    try {
      const raw = localStorage.getItem(getLockoutKey(p));
      if (!raw) return { isLocked: false, remainingSec: 0, attempts: 0 };
      const data = JSON.parse(raw);
      if (data.lockedUntil && Date.now() < data.lockedUntil) {
        const remainingSec = Math.ceil((data.lockedUntil - Date.now()) / 1000);
        return { isLocked: true, remainingSec, attempts: data.attempts || 5 };
      }
      if (data.lockedUntil && Date.now() >= data.lockedUntil) {
        localStorage.removeItem(getLockoutKey(p));
      }
      return { isLocked: false, remainingSec: 0, attempts: data.attempts || 0 };
    } catch(e) {
      return { isLocked: false, remainingSec: 0, attempts: 0 };
    }
  };

  const handleLogin = async (e) => {
    e.preventDefault();

    const cleanPhone = normalizePhone(phone);
    const lockStatus = checkLockout(cleanPhone);
    if (lockStatus.isLocked) {
      const mins = Math.ceil(lockStatus.remainingSec / 60);
      const msg = lang === 'en'
        ? `⚠️ Account is temporarily locked after 5 failed attempts. Please wait ${mins} minute(s) before trying again.`
        : `⚠️ تم إيقاف محاولات تسجيل الدخول مؤقتاً لحماية أمان حسابك بعد 5 محاولات غير صحيحة. يرجى الانتظار ${mins} دقيقة والمحاولة لاحقاً.`;
      setError(msg);
      alert(msg);
      return;
    }

    if (!password) return setError('يرجى إدخال كلمة المرور');
    setError('');
    setLoading(true);

    try {
      const custRef = ref(db, `PublicCustomers/${cleanPhone}`);
      const custSnap = await get(custRef);
      
      const internalEmail = getInternalEmail(cleanPhone);
      const dbEmail = custSnap.exists() && custSnap.val().Email ? custSnap.val().Email.trim() : null;
      const authEmail = custSnap.exists() && custSnap.val().AuthEmail ? custSnap.val().AuthEmail.trim() : null;
      
      const emailToTry1 = authEmail || internalEmail;
      
      try {
        await signInWithEmailAndPassword(auth, emailToTry1, password);
      } catch (firstErr) {
        if (dbEmail && dbEmail !== emailToTry1 && dbEmail.includes('@')) {
          try {
            await signInWithEmailAndPassword(auth, dbEmail, password);
          } catch (secondErr) {
            throw secondErr;
          }
        } else {
          throw firstErr;
        }
      }
      
      try { localStorage.removeItem(getLockoutKey(cleanPhone)); } catch(e) {}
      await refreshCustomerData(cleanPhone);
      setStep('dashboard');
    } catch (err) {
      let updatedAttempts = 1;
      try {
        const key = getLockoutKey(cleanPhone);
        const raw = localStorage.getItem(key);
        const data = raw ? JSON.parse(raw) : { attempts: 0 };
        data.attempts = (data.attempts || 0) + 1;
        data.lastAttempt = Date.now();
        if (data.attempts >= 5) {
          data.lockedUntil = Date.now() + 5 * 60 * 1000;
          localStorage.setItem(key, JSON.stringify(data));
          const lockMsg = lang === 'en'
            ? '⚠️ You have entered an incorrect password 5 times. Account login is locked for 5 minutes.'
            : '⚠️ لقد قمت بإدخال كلمة مرور غير صحيحة 5 مرات متتالية. تم إيقاف محاولات الدخول مؤقتاً لمدة 5 دقائق لحماية أمان حسابك.';
          setError(lockMsg);
          alert(lockMsg);
          setLoading(false);
          return;
        } else {
          localStorage.setItem(key, JSON.stringify(data));
          updatedAttempts = data.attempts;
        }
      } catch(e) {}

      const remainingAttempts = Math.max(0, 5 - updatedAttempts);
      const errMsg = lang === 'en'
        ? `Incorrect password. Remaining attempts before temporary lockout: ${remainingAttempts}`
        : `رقم الهاتف أو كلمة المرور غير صحيحة. المحاولات المتبقية قبل إيقاف الحساب مؤقتاً: ${remainingAttempts}`;
      setError(errMsg);
    }
    setLoading(false);
  };

  const handleSetPasswordForExisting = async (e) => {
    e.preventDefault();
    if (newPassword.length < 6) return setError('كلمة المرور يجب أن تكون 6 أحرف على الأقل');
    setError('');
    setLoading(true);

    const cleanPhone = normalizePhone(phone);
    try {
      const emailToUse = getInternalEmail(cleanPhone);
      let userRes;
      try {
        userRes = await createUserWithEmailAndPassword(auth, emailToUse, newPassword);
      } catch (authErr) {
        if (authErr.code === 'auth/email-already-in-use') {
          // Update credentials with unique timestamp email
          const altEmail = `${cleanPhone}_${Date.now()}@internal.theblackbox.com`;
          userRes = await createUserWithEmailAndPassword(auth, altEmail, newPassword);
        } else {
          throw authErr;
        }
      }

      await set(ref(db, `UidToPhone/${userRes.user.uid}`), cleanPhone);
      await update(ref(db, `PublicCustomers/${cleanPhone}`), {
        uid: userRes.user.uid,
        AuthEmail: userRes.user.email
      });

      await refreshCustomerData(cleanPhone);
      setStep('dashboard');
    } catch (err) {
      setError('حدث خطأ أثناء ضبط كلمة المرور: ' + err.message);
    }
    setLoading(false);
  };

  const handleCompleteRegistration = async (e) => {
    e.preventDefault();
    if (!name.trim()) return setError('يرجى إدخال اسمك');
    if (!selectedZone) return setError('يرجى اختيار منطقتك السكنية');
    if (password.length < 6) return setError('كلمة المرور يجب أن تكون 6 أحرف على الأقل');
    setError('');
    setLoading(true);

    const cleanPhone = normalizePhone(phone);
    try {
      const emailToUse = email.trim() ? email.trim() : getInternalEmail(cleanPhone);
      let res;
      try {
        res = await createUserWithEmailAndPassword(auth, emailToUse, password);
      } catch (authErr) {
        if (authErr.code === 'auth/email-already-in-use') {
          const altEmail = `${cleanPhone}_${Date.now()}@internal.theblackbox.com`;
          res = await createUserWithEmailAndPassword(auth, altEmail, password);
        } else {
          throw authErr;
        }
      }
      
      await set(ref(db, `UidToPhone/${res.user.uid}`), cleanPhone);
      await set(ref(db, `PublicCustomers/${cleanPhone}`), {
        Name: name.trim(),
        Phone: cleanPhone,
        Email: email.trim() || null,
        AuthEmail: res.user.email,
        Address: address.trim() || null,
        Zone: selectedZone,
        zone: selectedZone,
        uid: res.user.uid,
        RegisteredDate: new Date().toISOString(),
        Points: 50, // Welcome gift points
        TotalOrders: 0
      });

      await refreshCustomerData(cleanPhone);
      setStep('dashboard');
    } catch (err) {
      setError('حدث خطأ أثناء التسجيل: ' + err.message);
    }
    setLoading(false);
  };

  const handleForgotPassword = async () => {
    setError('');
    setLoading(true);
    try {
      const cleanPhone = normalizePhone(phone);
      await sendWhatsAppOtp(cleanPhone);
      setIsExistingCustomer(true);
      setStep('otp_verify');
      alert(lang === 'en' 
        ? 'A 6-digit verification code was sent to your WhatsApp to reset your password.' 
        : 'تم إرسال كود التحقق المكون من 6 أرقام على واتساب لإعادة تعيين كلمة المرور.');
    } catch (err) {
      setError('فشل إرسال كود الاستعادة: ' + err.message);
    }
    setLoading(false);
  };

  const handleLogout = () => {
    signOut(auth);
    setStep('phone');
    setPhone('');
    setPassword('');
    setOtpInput('');
  };
